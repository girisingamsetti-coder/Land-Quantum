    23→function statusColor(s: string) {
    24→  if (['Approved','Completed','Paid','Compliant','Resolved','Issued','Executed','Registered'].includes(s)) return 'bg-emerald-100 text-emerald-700 border-emerald-200'
    25→  if (['Under Review','In Progress','Pending','Open','Draft','Submitted','Notice Issued','Decision Made','Scheduled','Under Scrutiny'].includes(s)) return 'bg-amber-100 text-amber-700 border-amber-200'
    26→  if (['Rejected','Failed','Non-Compliant','Overdue','Cancelled'].includes(s)) return 'bg-red-100 text-red-700 border-red-200'
    27→  if (['Deferred','On Hold','Delayed','At Risk','Partially Paid'].includes(s)) return 'bg-orange-100 text-orange-700 border-orange-200'
    28→  return 'bg-slate-100 text-slate-600 border-slate-200'
    29→}
    30→
    31→function severityColor(s: string) {
    32→  if (s === 'Critical') return 'bg-red-100 text-red-700 border-red-200'
    33→  if (s === 'High') return 'bg-orange-100 text-orange-700 border-orange-200'
    34→  if (s === 'Medium') return 'bg-amber-100 text-amber-700 border-amber-200'
    35→  return 'bg-blue-100 text-blue-700 border-blue-200'
    36→}
    37→
    38→function severityTileColor(s: string) {
    39→  if (s === 'Critical') return { bg: 'bg-red-50', border: 'border-red-100', text: 'text-red-700', iconBg: 'bg-red-100', iconText: 'text-red-600' }
    40→  if (s === 'High') return { bg: 'bg-orange-50', border: 'border-orange-100', text: 'text-orange-700', iconBg: 'bg-orange-100', iconText: 'text-orange-600' }
    41→  if (s === 'Medium') return { bg: 'bg-amber-50', border: 'border-amber-100', text: 'text-amber-700', iconBg: 'bg-amber-100', iconText: 'text-amber-600' }
    42→  return { bg: 'bg-blue-50', border: 'border-blue-100', text: 'text-blue-700', iconBg: 'bg-blue-100', iconText: 'text-blue-600' }
    43→}
